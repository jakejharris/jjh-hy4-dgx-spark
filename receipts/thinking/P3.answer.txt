```go
// limiter.go
package ratelimit

import (
	"errors"
	"sync"
	"time"
)

// Clock supplies the current time. Tests can provide a deterministic fake.
type Clock interface {
	Now() time.Time
}

// SystemClock is the default production Clock backed by time.Now.
type SystemClock struct{}

func (SystemClock) Now() time.Time { return time.Now() }

// Limiter is a concurrency-safe token bucket.
// It starts full and refills one token every rate duration, up to burst tokens.
type Limiter struct {
	mu       sync.Mutex
	clock    Clock
	rate     time.Duration
	burst    int64
	tokens   int64
	last     time.Time
}

// New returns a full token bucket. rate must be positive, burst must be
// positive, and clock must be non-nil.
func New(rate time.Duration, burst int64, clock Clock) (*Limiter, error) {
	if rate <= 0 {
		return nil, errors.New("ratelimit: rate must be positive")
	}
	if burst <= 0 {
		return nil, errors.New("ratelimit: burst must be positive")
	}
	if clock == nil {
		return nil, errors.New("ratelimit: clock must not be nil")
	}
	return &Limiter{
		clock:  clock,
		rate:   rate,
		burst:  burst,
		tokens: burst,
		last:   clock.Now(),
	}, nil
}

// Allow reports whether one token is available and consumes it if so.
func (l *Limiter) Allow() bool {
	return l.AllowN(1)
}

// AllowN reports whether n tokens are available and consumes them if so.
// It returns false for n <= 0 or n > burst.
func (l *Limiter) AllowN(n int64) bool {
	if n <= 0 {
		return false
	}

	l.mu.Lock()
	defer l.mu.Unlock()

	if n > l.burst {
		return false
	}

	now := l.clock.Now()
	if now.Before(l.last) {
		l.last = now
	} else if now.After(l.last) {
		elapsed := now.Sub(l.last)
		add := int64(elapsed / l.rate)
		rem := elapsed % l.rate

		if add >= l.burst-l.tokens {
			l.tokens = l.burst
			l.last = now
		} else if add > 0 {
			l.tokens += add
			l.last = now.Add(-rem)
		}
	}

	if l.tokens < n {
		return false
	}
	l.tokens -= n
	return true
}
```

```go
// limiter_test.go
package ratelimit

import (
	"sync"
	"testing"
	"time"
)

type fakeClock struct {
	mu sync.Mutex
	t  time.Time
}

func newFakeClock() *fakeClock {
	return &fakeClock{t: time.Unix(0, 0).UTC()}
}

func (c *fakeClock) Now() time.Time {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.t
}

func (c *fakeClock) Advance(d time.Duration) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.t = c.t.Add(d)
}

func (c *fakeClock) Set(t time.Time) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.t = t
}

func TestNewValidation(t *testing.T) {
	clock := newFakeClock()
	tests := []struct {
		name  string
		rate  time.Duration
		burst int64
		clock Clock
	}{
		{"zero rate", 0, 1, clock},
		{"negative rate", -time.Second, 1, clock},
		{"zero burst", time.Second, 0, clock},
		{"negative burst", time.Second, -1, clock},
		{"nil clock", time.Second, 1, nil},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			if _, err := New(tt.rate, tt.burst, tt.clock); err == nil {
				t.Fatal("expected error")
			}
		})
	}
	if _, err := New(time.Second, 1, clock); err != nil {
		t.Fatalf("unexpected error: %v", err)
	}
}

func TestAllowN(t *testing.T) {
	tests := []struct {
		name    string
		rate    time.Duration
		burst   int64
		steps   []struct {
			advance time.Duration
			n       int64
			want    bool
		}
	}{
		{
			name:  "starts full and refills by rate",
			rate:  time.Second,
			burst: 2,
			steps: []struct {
				advance time.Duration
				n       int64
				want    bool
			}{
				{0, 1, true},
				{0, 1, true},
				{0, 1, false},
				{time.Second, 1, true},
				{0, 1, false},
				{2 * time.Second, 2, true},
				{0, 1, false},
			},
		},
		{
			name:  "fractional elapsed time is preserved",
			rate:  100 * time.Millisecond,
			burst: 1,
			steps: []struct {
				advance time.Duration
				n       int64
				want    bool
			}{
				{0, 1, true},
				{60 * time.Millisecond, 1, false},
				{40 * time.Millisecond, 1, true},
			},
		},
		{
			name:  "n greater than burst never succeeds",
			rate:  time.Millisecond,
			burst: 3,
			steps: []struct {
				advance time.Duration
				n       int64
				want    bool
			}{
				{0, 4, false},
				{time.Hour, 4, false},
				{0, 3, true},
			},
		},
		{
			name:  "invalid n is rejected",
			rate:  time.Second,
			burst: 2,
			steps: []struct {
				advance time.Duration
				n       int64
				want    bool
			}{
				{0, 0, false},
				{0, -1, false},
				{0, 2, true},
			},
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			clock := newFakeClock()
			lim, err := New(tt.rate, tt.burst, clock)
			if err != nil {
				t.Fatalf("New: %v", err)
			}
			for i, step := range tt.steps {
				clock.Advance(step.advance)
				if got := lim.AllowN(step.n); got != step.want {
					t.Fatalf("step %d: AllowN(%d) = %v, want %v", i, step.n, got, step.want)
				}
			}
		})
	}
}

func TestAllowNClockMovesBackwards(t *testing.T) {
	clock := newFakeClock()
	lim, err := New(time.Second, 1, clock)
	if err != nil {
		t.Fatalf("New: %v", err)
	}
	if !lim.AllowN(1) {
		t.Fatal("initial token should be allowed")
	}
	clock.Advance(time.Second)
	clock.Set(clock.Now().Add(-2 * time.Second))
	if lim.AllowN(1) {
		t.Fatal("token should not be available after clock moved backwards")
	}
	clock.Advance(time.Second)
	if !lim.AllowN(1) {
		t.Fatal("token should be available after refill from reset time")
	}
}

func TestAllowNConcurrent(t *testing.T) {
	clock := newFakeClock()
	lim, err := New(time.Hour, 100, clock)
	if err != nil {
		t.Fatalf("New: %v", err)
	}

	const goroutines = 200
	var wg sync.WaitGroup
	var mu sync.Mutex
	allowed := 0
	for i := 0; i < goroutines; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			if lim.AllowN(1) {
				mu.Lock()
				allowed++
				mu.Unlock()
			}
		}()
	}
	wg.Wait()

	if allowed != 100 {
		t.Fatalf("allowed = %d, want 100", allowed)
	}
}
```
