```go
// limiter.go
package ratelimit

import (
    "errors"
    "math"
    "reflect"
    "sync"
    "time"
)

type Clock interface {
    Now() time.Time
}

type SystemClock struct{}
func (SystemClock) Now() time.Time { return time.Now() }

var (
    ErrInvalidRate = errors.New("ratelimit: rate must be finite and positive")
    ErrInvalidBurst = errors.New("ratelimit: burst must be positive")
    ErrNilClock = errors.New("ratelimit: clock must not be nil")
)

type Limiter struct {
    clock Clock
    rate float64
    burst int

    mu sync.Mutex
    tokens float64
    last time.Time
}

func New(rate float64, burst int) (*Limiter, error) {
    return NewWithClock(rate, burst, SystemClock{})
}

func NewWithClock(rate float64, burst int, clock Clock) (*Limiter, error) {
    if math.IsNaN(rate) || math.IsInf(rate, 0) || rate <= 0 {
        return nil, ErrInvalidRate
    }
    if burst <= 0 {
        return nil, ErrInvalidBurst
    }
    if isNilClock(clock) {
        return nil, ErrNilClock
    }
    return &Limiter{
        clock: clock,
        rate: rate,
        burst: burst,
        tokens: float64(burst),
        last: clock.Now(),
    }, nil
}

func (l *Limiter) Allow() bool { return l.AllowN(1) }

func (l *Limiter) AllowN(n int) bool {
    if n <= 0 {
        return false
    }

    l.mu.Lock()
    defer l.mu.Unlock()

    if n > l.burst {
        return false
    }

    now := l.clock.Now()
    l.advance(now)

    need := float64(n)
    if l.tokens < need {
        return false
    }
    l.tokens -= need
    return true
}

func (l *Limiter) advance(now time.Time) {
    elapsed := now.Sub(l.last)
    if elapsed <= 0 {
        return
    }
    l.last = now

    added := elapsed.Seconds() * l.rate
    deficit := float64(l.burst) - l.tokens
    if math.IsNaN(added) || math.IsInf(added, 0) || added >= deficit {
        l.tokens = float64(l.burst)
        return
    }
    l.tokens += added
}

func isNilClock(c Clock) bool {
    if c == nil { return true }
    v := reflect.ValueOf(c)
    switch v.Kind() {
    case reflect.Chan, reflect.Func, reflect.Interface, reflect.Map, reflect.Ptr, reflect.Slice:
        return v.IsNil()
    default:
        return false
    }
}
```
